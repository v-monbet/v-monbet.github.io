Nearest neighbors method¶

For each value of the nearest neighbor numbers, Monte Carlo cross validation is run with aout 25% of the samples in the validation set. At each iteration de Mean Square Error is stored.

First block with uniform weights.

In [3]:
from sklearn import neighbors

nb_neighbors = [1,5,10,20]


B = 100 # number of samples for the cross validation
mse_knn= np.zeros((B,len(nb_neighbors)))
for b in range(0,B):
    X_train, X_test, y_train, y_test = train_test_split(X,fat,test_size=9)
    mk=np.mean(X_train,axis=0)
    sk=np.maximum(np.std(X_train,axis=0),10*np.finfo(float).eps)
    X_train, X_test = np.add(X_train,-mk), np.add(X_test,-mk)
    X_train, X_test = np.multiply(X_train,1/sk),np.multiply(X_test,1/sk)
    for i,k in enumerate(nb_neighbors):
        knn = neighbors.KNeighborsRegressor(k, weights='uniform')
        knn.fit(X_train,y_train)
        y_pred = knn.predict(X_test)
        mse_knn[b,i] = np.mean((y_pred-y_test)**2)
    
plt.boxplot(np.sqrt(mse_knn),labels=nb_neighbors)
plt.ylabel('Root mean square error')
plt.xlabel('Nb of neighbors')

print("Best RMSE of knn method: ",np.round(np.sqrt(np.min(np.mean(mse_knn,axis=0))),2))

Best RMSE of knn method:  0.06

Shuffle split generate the B training and validation sets.

from sklearn.model_selection import GridSearchCV, cross_val_score
from sklearn.model_selection import ShuffleSplit

# Generation of the B training/validation sets
cv = ShuffleSplit(n_splits=B, 
                  test_size=0.25, 
                  random_state=0)
# Initialization of the regressor
knn = neighbors.KNeighborsRegressor()
param_grid = [{'n_neighbors': [1,5,10,20], 
               'weights': ['uniform','distance'],
               'metric': ['euclidean','minkowski']}]
# Optimization of the hyperparameters
search = GridSearchCV(knn, 
                      param_grid, 
                      scoring="neg_mean_squared_error",
                      cv=cv,
                      n_jobs=-1)
search.fit(X,fat)
